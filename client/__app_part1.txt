const auth = asyncRoute(
  async (req: Authed, res: Response, next: NextFunction) => {
    const token = req.headers.authorization?.replace(/^Bearer\s+/, "");
    if (!token)
      throw Object.assign(new Error("Authentication required"), {
        status: 401,
      });
    let claims: any;
    try {
      claims = jwt.verify(token, secret());
    } catch {
      throw Object.assign(
        new Error("Invalid or expired authentication token"),
        { status: 401 },
      );
    }
    const user = await User.findById(claims.id);
    if (!user || user.status === "DISABLED")
      throw Object.assign(new Error("Account unavailable"), { status: 401 });
    req.user = { id: user.id, role: user.role as ResourceRole };
    next();
  },
);

app.post(
  "/api/v1/auth/register",
  asyncRoute(async (req, res) => {
    const input = z
      .object({
        name: z.string().min(2),
        email: z
          .string()
          .email()
          .transform((v) => v.toLowerCase()),
        password: z.string().min(8),
        role: z.enum(["PLAYER", "COACH"]).default("PLAYER"),
      })
      .parse(req.body);
    if (await User.exists({ email: input.email }))
      throw Object.assign(new Error("Email already registered"), {
        status: 409,
      });
    const user = await User.create({
      ...input,
      passwordHash: await bcrypt.hash(input.password, 12),
      status: input.role === "COACH" ? "PENDING" : "ACTIVE",
    });
    if (input.role === "PLAYER") await Player.create({ user: user._id });
    else await Coach.create({ user: user._id, approved: false });
    const token = jwt.sign({ id: user.id, role: user.role }, secret(), {
      expiresIn: "7d",
    });
    res
      .status(201)
      .json({
        token,
        user: {
          id: user.id,
          name: user.name,
          email: user.email,
          role: user.role,
        },
      });
  }),
);

app.post(
  "/api/v1/subscriptions",
  ...roles("PLAYER"),
  asyncRoute(async (req: Authed, res) => {
    const { planId } = z
      .object({ planId: z.string().regex(/^[a-f\d]{24}$/i) })
      .parse(req.body);
    const plan = await Plan.findOne({ _id: planId, published: true });
    if (!plan)
      throw Object.assign(new Error("Plan unavailable"), { status: 404 });
    await releaseExpiredSubscriptions();
    const now = new Date();
    const legacy = await Subscription.find({
      player: req.user!.id,
      activeSlot: { $ne: true },
    }).select("status expiresAt");
    if (
      legacy.some((item) =>
        hasBlockingSubscription(item.status, item.expiresAt, now),
      ) ||
      (await Subscription.exists({ player: req.user!.id, activeSlot: true }))
    )
      throw Object.assign(
        new Error("An active or pending subscription already exists"),
        { status: 409 },
      );
    let sub;
    try {
      sub = await Subscription.create({
        player: req.user!.id,
        activeSlot: true,
        plan: plan._id,
        planSnapshot: {
          name: plan.name,
          pricePaise: plan.pricePaise,
          durationDays: plan.durationDays,
          maxVideos: plan.maxVideos,
          includedReviews: plan.includedReviews,
          features: plan.features,
        },
      });
    } catch (error: any) {
      if (error.code === 11000)
        throw Object.assign(
          new Error("An active or pending subscription already exists"),
          { status: 409 },
        );
      throw error;
    }
    try {
      let payment;
      if (process.env.RAZORPAY_KEY_ID && process.env.RAZORPAY_KEY_SECRET) {
        const order = await createRazorpayOrder(
          plan.pricePaise,
          sub.id,
          sub.id,
        );
        payment = await Payment.create({
          subscription: sub._id,
          amountPaise: plan.pricePaise,
          orderId: order.id,
        });
        res
          .status(201)
          .json({
            subscription: sub,
            payment,
            checkout: {
              enabled: true,
              keyId: process.env.RAZORPAY_KEY_ID,
              orderId: order.id,
              amount: order.amount,
              currency: order.currency,
            },
          });
        return;
      }
      if (process.env.NODE_ENV === "production")
        throw Object.assign(
          new Error("Razorpay credentials are required in production"),
          { status: 503 },
        );
      payment = await Payment.create({
        subscription: sub._id,
        amountPaise: plan.pricePaise,
      });
      res
        .status(201)
        .json({
          subscription: sub,
          payment,
          checkout: { enabled: false, keyId: null },
        });
    } catch (error) {
      await Subscription.deleteOne({ _id: sub._id });
      throw error;
    }
  }),
);
app.post(
  "/api/v1/subscriptions/:id/payment-order",
  ...roles("PLAYER"),
  asyncRoute(async (req: Authed, res) => {
    const sub = await Subscription.findOneAndUpdate(
      {
        _id: req.params.id,
        player: req.user!.id,
        status: "PENDING_PAYMENT",
        paymentOrderPending: { $ne: true },
      },
      { $set: { paymentOrderPending: true } },
      { new: true },
    );
    if (!sub)
      throw Object.assign(
        new Error(
          "Pending subscription not found or payment order is already being created",
        ),
        { status: 409 },
      );
    try {
      const snapshot = sub.planSnapshot;
      if (!snapshot) throw new Error("Subscription plan snapshot is missing");
      if (!process.env.RAZORPAY_KEY_ID || !process.env.RAZORPAY_KEY_SECRET)
        throw Object.assign(
          new Error("Razorpay credentials are not configured"),
          { status: 503 },
        );
      const existing = await Payment.findOne({
        subscription: sub._id,
        state: { $in: ["CREATED", "PENDING"] },
        orderId: { $exists: true },
      });
      if (existing) {
        await Subscription.updateOne(
          { _id: sub._id },
          { $set: { paymentOrderPending: false } },
        );
        res.json({
          payment: existing,
          checkout: {
            enabled: true,
            keyId: process.env.RAZORPAY_KEY_ID,
            orderId: existing.orderId,
            amount: existing.amountPaise,
            currency: existing.currency,
          },
        });
        return;
      }
      const order = await createRazorpayOrder(
        snapshot.pricePaise!,
        sub.id,
        sub.id,
      );
      const payment = await Payment.create({
        subscription: sub._id,
        amountPaise: snapshot.pricePaise,
        orderId: order.id,
      });
      await Subscription.updateOne(
        { _id: sub._id },
        { $set: { paymentOrderPending: false } },
      );
      res
        .status(201)
        .json({
          payment,
          checkout: {
            enabled: true,
            keyId: process.env.RAZORPAY_KEY_ID,
            orderId: order.id,
            amount: order.amount,
            currency: order.currency,
          },
        });
    } catch (error) {
      await Subscription.updateOne(
        { _id: sub._id },
        { $set: { paymentOrderPending: false } },
      );
      throw error;
    }
  }),
);

app.post(
  "/api/v1/payments/:id/verify",
  ...roles("PLAYER"),
  asyncRoute(async (req: Authed, res) => {
    const payment = await Payment.findById(req.params.id).populate(
      "subscription",
    );
    if (!payment)
      throw Object.assign(new Error("Payment not found"), { status: 404 });
    const sub: any = payment.subscription;
    if (String(sub.player) !== req.user!.id)
      throw Object.assign(new Error("Forbidden"), { status: 403 });
    const input = z
      .object({
        razorpay_payment_id: z.string().min(1),
        razorpay_order_id: z.string().min(1),
        razorpay_signature: z.string().min(1),
      })
      .parse(req.body);
    if (input.razorpay_order_id !== payment.orderId)
      throw Object.assign(new Error("Payment order does not match"), {
        status: 400,
      });
    const result = await reconcilePayment(
      input.razorpay_payment_id,
      input.razorpay_order_id,
      input.razorpay_signature,
    );
    if (result.changed)
      await notice(
        req.user!.id,
        "Payment verified",
        "Choose your coach to start your coaching period.",
        "PAYMENT",
      );
    res.json({
      subscription: await Subscription.findById(sub._id),
      payment: result.payment,
    });
  }),
);
